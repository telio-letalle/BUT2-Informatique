NOMCAMPING SALAIRE MOYEN

La Décharge Monochrome 1212.5

SELECT c.nomCamping, AVG(salaireEmploye) FROM CAMPINGS c JOIN EMPLOYES e ON c.idcamping=e.idcamping GROUP BY c.nomCamping;

La Décharge Monochrome

SELECT nomCamping FROM CAMPINGS c JOIN EMPLOYES e ON c.idcamping=e.idcamping GROUP BY c.nomCamping HAVING COUNT(*)>3
