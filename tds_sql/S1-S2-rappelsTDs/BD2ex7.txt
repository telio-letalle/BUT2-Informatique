R100 : le nom des bungalows qui proposent tous les services (c'est-à-dire tous les services qui sont dans la table Services). Réaliser cette division d’au moins deux façons différentes
NOMBUNGALOW
------------
Le Palace

SELECT nomBungalow FROM Bungalows b JOIN Proposer p ON b.idBungalow=p.idBungalow Group BY b.idBungalow, nomBungalow HAVING COUNT(*) = (SELECT COUNT(*) FROM SERVICES);

SELECT nomBungalow FROM Bungalows b
WHERE NOT EXISTS (
SELECT idService FROM Services
MINUS
SELECT idService FROM Proposer p
WHERE p.idBungalow= b.idBungalow);


R101 : le nom des bungalows qui proposent tous les services de la catégorie ‘Luxe’.
Réaliser cette division d’au moins deux façons différentes
NOMBUNGALOW
------------------
Le Palace
Le Titanic
La Suite Nuptiale

SELECT nomBungalow FROM Bungalows b JOIN Proposer p ON b.idBungalow=p.idBungalow JOIN Services s ON s.idService=p.idService Where categorieService = 'Luxe' Group BY b.idBungalow, nomBungalow HAVING COUNT(*) = (SELECT COUNT(*) FROM SERVICES Where categorieService = 'Luxe');

SELECT nomBungalow FROM Bungalows b
WHERE NOT EXISTS (
SELECT idService FROM Services Where categorieService = 'Luxe'
MINUS
SELECT s.idService FROM Proposer p JOIN Services s ON s.idService=p.idService 
WHERE categorieService = 'Luxe' and p.idBungalow= b.idBungalow);
