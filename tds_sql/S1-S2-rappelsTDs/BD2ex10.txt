CREATE OR REPLACE VIEW BUNGALOWSLFB (idBungalow, nomBungalow, superficieBungalow) AS
SELECT idBungalow, nomBungalow, superficieBungalow FROM BUNGALOWS b JOIN CAMPINGS c ON C.idCamping = b.idCamping WHERE nomCamping = 'Les Flots Bleus'

CREATE OR REPLACE VIEW LOCATIONSBUNGALOWS (idBungalow, nomBungalow, idCamping, nomCamping, superficieBungalow, nbLocations, CA, CAML) AS
SELECT b.idBungalow, nomBungalow, b.idCamping, nomCamping, superficieBungalow, COUNT(*) AS nbLocations, SUM(montantLocation) AS CA,AVG(Distinct montantLocation) AS CAML FROM BUNGALOWS b JOIN Campings c ON c.idCamping= b.idCamping JOIN Locations l ON l.idBungalow=b.idBungalow GROUP BY b.idBungalow, nomBungalow, b.idCamping, nomCamping, superficieBungalow;
